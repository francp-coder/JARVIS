async function askJarvis(question) { setMode("thinking"); addLogEntry("user", question); els.interimText.textContent = ""; if (!WORKER_URL || WORKER_URL.indexOf("INCOLLA_QUI") !== -1) { const msg = "Non ho ancora un indirizzo del server configurato. Segui le istruzioni per impostare WORKER_URL."; addLogEntry("jarvis", msg); speak(msg); return; } try { const response = await fetch(WORKER_URL, { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ question }) }); const data = await response.json(); // ==================================================== // GESTIONE DEI LINK (Inclusi App Deep Links come discord://) // ==================================================== let linksToOpen = []; let rawAnswer = (data.text || "Non sono riuscito a elaborare una risposta.").trim(); // 1. Regex modificata per accettare protocolli come discord://, spotify:// oltre a http/https let htmlAnswer = rawAnswer.replace(/\[([^\]]+)\]\s*\(\s*([a-zA-Z0-9.\-]+:\/\/[^\s\)]*)\s*\)/g, function(match, testo, url) { linksToOpen.push(url); return `${testo}`; }); // 2. Togli gli URL grezzi se non sono tra parentesi htmlAnswer = htmlAnswer.replace(/(?)(https?:\/\/[^\s\)<>]+)/g, function(match, url) { let cleanUrl = url.replace(/[\.,;:!]+$/, ''); if (!linksToOpen.includes(cleanUrl)) { linksToOpen.push(cleanUrl); } return `${cleanUrl}`; }); // 3. Prepara il testo per la voce (togliendo i link) let speechAnswer = rawAnswer.replace(/\[([^\]]+)\]\s*\(\s*([a-zA-Z0-9.\-]+:\/\/[^\s\)]*)\s*\)/g, "$1"); speechAnswer = speechAnswer.replace(/(https?:\/\/[^\s\)<>]+)/g, ""); speechAnswer = speechAnswer.replace(/\*/g, "").replace(/\n/g, " ").trim(); // 4. Mettiamo i tag
per andare a capo nella chat visiva htmlAnswer = htmlAnswer.replace(/\n/g, '
'); // ==================================================== // AZIONI FINALI // ==================================================== // Tenta l'apertura automatica (I browser potrebbero bloccare discord:// al primo tentativo per sicurezza) linksToOpen.forEach(url => { // Usa window.location per i deep link delle app, window.open per i siti web if (url.startsWith('http')) { window.open(url, '_blank'); } else { // I deep link a volte vengono bloccati dal popup blocker o interpretati male se usati con _blank window.location.href = url; } }); // Mostra in chat passando isHTML = true per far renderizzare i pulsanti addLogEntry("jarvis", htmlAnswer, true); // Fai parlare Jarvis speak(speechAnswer); } catch (e) { const msg = "Ho avuto un problema a contattare il server. Riprova."; addLogEntry("jarvis", msg); speak(msg); } }
